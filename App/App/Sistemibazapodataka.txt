CREATE TABLE OGLAS (
    OGLAS_ID NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    NAZIV_POZICIJE VARCHAR2(200) NOT NULL,
    VRSTA_OGLASA VARCHAR2(20) NOT NULL
        CONSTRAINT CHK_VRSTA_OGLASA
        CHECK (
            VRSTA_OGLASA IN (
                'STALNI',
                'PRIVREMENI',
                'PRAKSA',
                'SEZONSKI'
            )
        ),
    OPIS CLOB,
    ZAHTEVI CLOB,
    MIN_PLATA NUMBER(10,2),
    MAX_PLATA NUMBER(10,2),
    DATUM_OBJAVE DATE NOT NULL,
    DATUM_ZATVARANJA DATE,
    STATUS VARCHAR2(30) NOT NULL
        CONSTRAINT CHK_STATUS_OGLAS
        CHECK (
            STATUS IN (
                'AKTIVAN',
                'ZATVOREN',
                'U_TOKU_SELEKCIJE'
            )
        ),

    CONSTRAINT CHK_PLATA
        CHECK (
            MIN_PLATA IS NULL
            OR MAX_PLATA IS NULL
            OR MIN_PLATA <= MAX_PLATA
        ),

    CONSTRAINT CHK_DATUMI_OGLAS
        CHECK (
            DATUM_ZATVARANJA IS NULL
            OR DATUM_ZATVARANJA > DATUM_OBJAVE
        )
);


CREATE TABLE OGLAS_PRIVREMENI (
    OGLAS_ID NUMBER PRIMARY KEY,
    PROJEKAT VARCHAR2(300) NOT NULL,
    PERIOD_ANGAZOVANJA VARCHAR2(100) NOT NULL,

    CONSTRAINT FK_PRIVREMENI_OGLAS
        FOREIGN KEY (OGLAS_ID)
        REFERENCES OGLAS(OGLAS_ID)
        ON DELETE CASCADE
);


CREATE OR REPLACE TRIGGER TRG_PRIVREMENI_TIP
BEFORE INSERT OR UPDATE ON OGLAS_PRIVREMENI
FOR EACH ROW
DECLARE
    V_VRSTA OGLAS.VRSTA_OGLASA%TYPE;
BEGIN
    SELECT VRSTA_OGLASA
    INTO V_VRSTA
    FROM OGLAS
    WHERE OGLAS_ID = :NEW.OGLAS_ID;

    IF V_VRSTA <> 'PRIVREMENI' THEN
        RAISE_APPLICATION_ERROR(
            -20001,
            'Oglas mora biti tipa PRIVREMENI.'
        );
    END IF;
END;
/


CREATE TABLE OGLAS_PRAKSA (
    OGLAS_ID NUMBER PRIMARY KEY,
    IME_ZAPOSLENOG VARCHAR2(100) NOT NULL,
    PREZIME_ZAPOSLENOG VARCHAR2(100) NOT NULL,
    DUZINA_TRAJANJA NUMBER(2) NOT NULL
        CONSTRAINT CHK_DUZINA_TRAJANJA

    CONSTRAINT FK_PRAKSA_OGLAS

CREATE TABLE OGLAS_SEZONSKI (
    OGLAS_ID NUMBER PRIMARY KEY,
    SEZONA VARCHAR2(100) NOT NULL,

    CONSTRAINT FK_SEZONSKI_OGLAS
        FOREIGN KEY (OGLAS_ID)

CREATE TABLE CV (
    CV_ID NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    OGLAS_ID NUMBER NOT NULL,
    IME VARCHAR2(100) NOT NULL,
    PREZIME VARCHAR2(100) NOT NULL,
    EMAIL VARCHAR2(200) NOT NULL,
    TELEFON VARCHAR2(30),
    DATUM_PODNOSENJA DATE NOT NULL,
    STATUS VARCHAR2(30) NOT NULL
        CONSTRAINT CHK_STATUS_CV
        CHECK (
            STATUS IN (
                'PRIMLJEN',
                'U_PROCESU',
                'ODBIJEN',
                'POZVAN_NA_INTERVJU'
            )
        ),

    CONSTRAINT FK_CV_OGLAS
        FOREIGN KEY (OGLAS_ID)
        REFERENCES OGLAS(OGLAS_ID),

    CONSTRAINT UQ_EMAIL_OGLAS
        UNIQUE (EMAIL, OGLAS_ID)
);


CREATE TABLE INTERVJU (
    INTERVJU_ID NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    CV_ID NUMBER NOT NULL,
    DATUM DATE NOT NULL,
    VREME DATE NOT NULL,
    TIP VARCHAR2(20) NOT NULL
        CONSTRAINT CHK_TIP_INTERVJU
        CHECK (
            TIP IN (
                'LICNI',
                'VIDEO',
                'TELEFONSKI'
            )
        ),
    LOKACIJA VARCHAR2(300),
    IME_ZAPOSLENOG VARCHAR2(100) NOT NULL,
    PREZIME_ZAPOSLENOG VARCHAR2(100) NOT NULL,
    OCENA NUMBER(2)
        CONSTRAINT CHK_OCENA
        CHECK (OCENA BETWEEN 1 AND 10),
    NAPOMENE CLOB,

    CONSTRAINT FK_INTERVJU_CV
        FOREIGN KEY (CV_ID)
        REFERENCES CV(CV_ID)
        ON DELETE CASCADE
);


CREATE TABLE TEST (
    TEST_ID NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    CV_ID NUMBER NOT NULL,
    REZULTAT NUMBER(5,2)
        CONSTRAINT CHK_REZULTAT
        CHECK (REZULTAT BETWEEN 0 AND 100),
    DATUM_TESTIRANJA DATE NOT NULL,
    VRSTA_TESTIRANJA VARCHAR2(100) NOT NULL,
    KOMENTAR CLOB,

    CONSTRAINT FK_TEST_CV
        FOREIGN KEY (CV_ID)
        REFERENCES CV(CV_ID)
        ON DELETE CASCADE
);


CREATE TABLE ODLUKA (
    ODLUKA_ID NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    CV_ID NUMBER NOT NULL UNIQUE,
    STATUS VARCHAR2(20) NOT NULL
        CONSTRAINT CHK_STATUS_ODLUKA
        CHECK (
            STATUS IN (
                'IZABRAN',
                'ODBIJEN',
                'REZERVA',
                'NA_CEKANJU'
            )
        ),
    DATUM_DONOSENJA_ODLUKE DATE NOT NULL,
    PONUDJENA_PLATA NUMBER(10,2),
    PRIHVATANJE_PONUDE CHAR(1)
        CONSTRAINT CHK_PRIHVATANJE
        CHECK (
            PRIHVATANJE_PONUDE IN ('D', 'N')
        ),
    DATUM_POCETKA_RADA DATE,
    RAZLOG_ODBIJANJA VARCHAR2(500),

    CONSTRAINT FK_ODLUKA_CV
        FOREIGN KEY (CV_ID)
        REFERENCES CV(CV_ID)
        ON DELETE CASCADE
);


CREATE INDEX IDX_CV_OGLAS
    ON CV(OGLAS_ID);

CREATE INDEX IDX_CV_EMAIL
    ON CV(EMAIL);

CREATE INDEX IDX_INTERVJU_CV
    ON INTERVJU(CV_ID);

CREATE INDEX IDX_TEST_CV
    ON TEST(CV_ID);

CREATE INDEX IDX_OGLAS_STATUS
    ON OGLAS(STATUS);
